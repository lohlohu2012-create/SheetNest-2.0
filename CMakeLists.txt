cmake_minimum_required(VERSION 3.21)
project(SheetNest2 VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SHEETNEST_BUILD_TESTS "Build tests" ON)
option(SHEETNEST_BUILD_DESKTOP "Build Qt desktop application" OFF)
add_subdirectory(core)
if(SHEETNEST_BUILD_TESTS)
 enable_testing()
 add_subdirectory(tests)
endif()
if(SHEETNEST_BUILD_DESKTOP)
 add_subdirectory(desktop)
endif()
